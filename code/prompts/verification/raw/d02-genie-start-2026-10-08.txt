Q: Which market has the highest quarter-over-quarter net profit growth?
status: COMPLETED
SQL:
WITH market_quarterly AS (
  SELECT
    gold_property_quarterly.market,
    gold_property_quarterly.fiscal_quarter,
    gold_property_quarterly.fiscal_quarter_date,
    SUM(gold_property_quarterly.net_profit) AS market_net_profit
  FROM thhart_october_2026_catalog.revantage_20261008.gold_property_quarterly
  WHERE gold_property_quarterly.market IS NOT NULL
    AND gold_property_quarterly.fiscal_quarter_date IS NOT NULL
    AND gold_property_quarterly.net_profit IS NOT NULL
  GROUP BY gold_property_quarterly.market, gold_property_quarterly.fiscal_quarter, gold_property_quarterly.fiscal_quarter_date
), market_growth AS (
  SELECT
    market,
    fiscal_quarter,
    fiscal_quarter_date,
    market_net_profit,
    LAG(market_net_profit) OVER (PARTITION BY market ORDER BY fiscal_quarter_date) AS prior_market_net_profit
  FROM market_quarterly
), ranked AS (
  SELECT
    market,
    fiscal_quarter,
    market_net_profit,
    prior_market_net_profit,
    100 * TRY_DIVIDE(market_net_profit - prior_market_net_profit, prior_market_net_profit) AS qoq_net_profit_growth_pct,
    RANK() OVER (
      ORDER BY 100 * TRY_DIVIDE(market_net_profit - prior_market_net_profit, prior_market_net_profit) DESC
    ) AS growth_rank
  FROM market_growth
  WHERE prior_market_net_profit IS NOT NULL
    AND market IS NOT NULL
    AND market_net_profit IS NOT NULL
    AND prior_market_net_profit IS NOT NULL
)
SELECT
  market,
  fiscal_quarter,
  market_net_profit,
  prior_market_net_profit,
  qoq_net_profit_growth_pct
FROM ranked
WHERE growth_rank = 1;

Q: Summarize the top 3 risks across the portfolio this quarter
status: COMPLETED
SQL:
WITH `base` AS (
  SELECT
    `property_id`,
    `property_name`,
    `region_name`,
    `market`,
    `net_profit`,
    `net_profit_target`,
    `qoq_change_pct`
  FROM `thhart_october_2026_catalog`.`revantage_20261008`.`gold_property_quarterly`
  WHERE `fiscal_quarter` = '2026-Q2'
    AND `property_name` IS NOT NULL
    AND `region_name` IS NOT NULL
    AND `market` IS NOT NULL
    AND `net_profit` IS NOT NULL
    AND `net_profit_target` IS NOT NULL
    AND `qoq_change_pct` IS NOT NULL
), `risks` AS (
  SELECT
    'Target miss gap' AS `risk_type`,
    SUM(CASE WHEN `net_profit` < `net_profit_target` THEN `net_profit_target` - `net_profit` ELSE 0 END) AS `risk_value`,
    CONCAT(CAST(SUM(CASE WHEN `net_profit` < `net_profit_target` THEN 1 ELSE 0 END) AS STRING), ' properties missed target') AS `risk_summary`
  FROM `base`

  UNION ALL

  SELECT
    'Quarter-over-quarter profit decline' AS `risk_type`,
    SUM(CASE WHEN `qoq_change_pct` < 0 THEN -`qoq_change_pct` ELSE 0 END) AS `risk_value`,
    CONCAT(CAST(SUM(CASE WHEN `qoq_change_pct` < 0 THEN 1 ELSE 0 END) AS STRING), ' properties had negative QoQ net profit change') AS `risk_summary`
  FROM `base`

  UNION ALL

  SELECT
    'Loss-making properties' AS `risk_type`,
    SUM(CASE WHEN `net_profit` < 0 THEN -`net_profit` ELSE 0 END) AS `risk_value`,
    CONCAT(CAST(SUM(CASE WHEN `net_profit` < 0 THEN 1 ELSE 0 END) AS STRING), ' properties had negative net profit') AS `risk_summary`
  FROM `base`
), `ranked` AS (
  SELECT
    `risk_type`,
    `risk_value`,
    `risk_summary`,
    RANK() OVER (ORDER BY `risk_value` DESC) AS `risk_rank`
  FROM `risks`
)
SELECT
  `risk_type`,
  `risk_value`,
  `risk_summary`
FROM `ranked`
WHERE `risk_rank` <= 3
ORDER BY `risk_rank`, `risk_type`
ROWS: ['risk_type', 'risk_value', 'risk_summary']
    ['Target miss gap', '156229.31', '3 properties missed target']
    ['Quarter-over-quarter profit decline', '44.05', '1 properties had negative QoQ net profit change']
    ['Loss-making properties', '0.00', '0 properties had negative net profit']
TEXT: This quarter, the biggest portfolio risk is **target misses**, with a combined shortfall of **EUR 156,229.31** across **3 properties**. The next risk is **quarter-over-quarter profit decline**, affecting **1 property** with a total decline of **44.05 percentage points**, while **no properties** were loss-making in **2026-Q2**. Across the 3 measured risk categories, **target misses stand out clearly as the main portfolio issue this quarter**.
